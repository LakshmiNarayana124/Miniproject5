        version: str
    ) -> Optional[dict[str, Any]]:
        """
        Return the manifest for ``(family, version)``.

        The manifest is read from the in-memory catalog when available,
        otherwise it is loaded from the on-disk ``manifest.json``.
        Returns ``None`` when the model is not registered.
        """

        family = str(family).strip()
        version = str(version).strip()

        with self._lock:
            entry = self._catalog.get((family, version))

        if entry is not None:
            return dict(entry["manifest"])

        # Fall back to the on-disk manifest.
        manifest_path = (
            self._models_dir / family / version / "manifest.json"
        )

        if manifest_path.exists():
            return json.loads(
                manifest_path.read_text(encoding="utf-8")
            )

        return None

    # -------------------------------------------------
    # LIST MODELS
    # -------------------------------------------------

    def list_models(
        self,
        family: Optional[str] = None
    ) -> list[dict[str, Any]]:
        """
        List registered models.

        When ``family`` is given, only models in that family are
        returned. Each item carries the manifest metadata and the
        active flag, but never the model object itself.
        """

        family = str(family).strip() if family else None

        with self._lock:
            entries = list(self._catalog.items())

        results: list[dict[str, Any]] = []

        for (model_family, model_version), entry in entries:
            if family and model_family != family:
                continue

            manifest = dict(entry["manifest"])
            manifest["is_active"] = entry["is_active"]
            results.append(manifest)

        return results

    # =====================================================
    # DATABASE HELPERS
    # =====================================================

    def _upsert_model_row(
        self,
        family: str,
        model_name: str,
        version: str,
        training_timestamp: str,
        dataset_hash: str,
        artifact_path: str,
        metrics: dict[str, Any]
    ) -> None:
        """Insert or update a ``ml_models`` row for the model."""

        connection = database.get_connection()

        try:
            connection.execute("""
                INSERT INTO ml_models (
                    family,
                    model_name,
                    version,
                    is_active,
                    training_timestamp,
                    dataset_hash,
                    artifact_path,
                    metrics_json,
                    schema_version
                )
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)

                ON CONFLICT(family, version)
                DO UPDATE SET
                    model_name = excluded.model_name,
                    training_timestamp = excluded.training_timestamp,
                    dataset_hash = excluded.dataset_hash,
                    artifact_path = excluded.artifact_path,
                    metrics_json = excluded.metrics_json,
                    schema_version = excluded.schema_version
            """, (
                family,
                model_name,
                version,
                0,
                training_timestamp,
                dataset_hash,
                artifact_path,
                json.dumps(metrics, default=str),
                SCHEMA_VERSION
            ))

            connection.commit()

        except Exception:
            connection.rollback()
            raise

        finally:
            connection.close()

    def _model_row_exists(self, family: str, version: str) -> bool:
        """Return ``True`` when a matching ``ml_models`` row exists."""

        connection = database.get_connection()

        try:
            row = connection.execute("""
                SELECT 1
                FROM ml_models
                WHERE family = ? AND version = ?
            """, (family, version)).fetchone()

            return row is not None

        finally:
            connection.close()

    def _deactivate_family_rows(self, family: str) -> None:
        """Set ``is_active = 0`` for every version in ``family``."""

        connection = database.get_connection()

        try:
            connection.execute("""
                UPDATE ml_models
                SET is_active = 0
                WHERE family = ?
            """, (family,))

            connection.commit()

        except Exception:
            connection.rollback()
            raise

        finally:
            connection.close()

    def _activate_model_row(self, family: str, version: str) -> None:
        """Set ``is_active = 1`` for the selected ``(family, version)``."""

        connection = database.get_connection()

        try:
            connection.execute("""
                UPDATE ml_models
                SET is_active = 1
                WHERE family = ? AND version = ?
            """, (family, version))

            connection.commit()

        except Exception:
            connection.rollback()
            raise

        finally:
            connection.close()

    def _active_version_from_db(self, family: str) -> Optional[str]:
        """Return the active version for ``family`` from the database."""

        connection = database.get_connection()

        try:
            row = connection.execute("""
                SELECT version
                FROM ml_models
                WHERE family = ? AND is_active = 1
                LIMIT 1
            """, (family,)).fetchone()

            return row["version"] if row else None

        finally:
            connection.close()

"""
INTELLIWATCH - OBSERVABILITY SERVICE
Model/system status responses and degradation alerting.
Requirements: 14.1-14.4
"""

import json
import logging
from datetime import datetime, timezone

import database
import model_registry as _mr

logger = logging.getLogger(__name__)

# Degradation threshold (Req 14.4): alert when metric drops >0.05 below target
DEGRADATION_MARGIN = 0.05
TABULAR_ACCURACY_TARGET = 0.92
UNSUPERVISED_F1_TARGET = 0.90

# Module-level current state (set by EquipmentController / IntelliWatchBrain)
_current_status = {"status": "NORMAL", "intent": "RUN", "timestamp": None}
_degradation_alerts = []


def set_current_status(status: str, intent: str):
    """Update the live system status (called by brain/controller)."""
    _current_status["status"] = status
    _current_status["intent"] = intent
    _current_status["timestamp"] = datetime.now(timezone.utc).isoformat()


# --------------------------------------------------------
# 11.1 — Model status response (Req 14.1)
# --------------------------------------------------------
def get_model_status(family: str = "ensemble"):
    """Return active model version, last training timestamp, metrics (Req 14.1)."""
    registry = _mr.registry
    model = registry.active_model(family)
    if model is None:
        return {"family": family, "active": False, "version": None, "metrics": {}}
    version = getattr(model, "version", "unknown")
    manifest = registry.get_manifest(family, version) or {}
    return {
        "family": family,
        "active": True,
        "version": version,
        "training_timestamp": manifest.get("training_timestamp"),
        "dataset_hash": manifest.get("dataset_hash"),
        "metrics": manifest.get("metrics", {}),
    }


def get_all_model_statuses():
    """Return status for all families."""
    families = ["tabular", "unsupervised", "sequence", "ensemble", "explorer"]
    return {f: get_model_status(f) for f in families}


# --------------------------------------------------------
# 11.1 — System status response (Req 14.2)
# --------------------------------------------------------
def get_system_status(iot_layer=None):
    """Return current system status, control intent, and device status (Req 14.2)."""
    devices = []
    if iot_layer is not None:
        try:
            devices = iot_layer.list_devices()
        except Exception:
            pass
    return {
        "system_status": _current_status.get("status", "NORMAL"),
        "control_intent": _current_status.get("intent", "RUN"),
        "timestamp": _current_status.get("timestamp"),
        "edge_devices": devices,
    }


# --------------------------------------------------------
# 11.2 — Degradation alerting (Req 14.4)
# --------------------------------------------------------
def check_model_degradation():
    """Raise alerts when deployed metrics fall >0.05 below target (Req 14.4)."""
    alerts = []
    for family in ["tabular", "unsupervised", "ensemble"]:
        info = get_model_status(family)
        if not info.get("active"):
            continue
        metrics = info.get("metrics", {})
        if family == "unsupervised":
            f1 = metrics.get("f1", 1.0)
            if f1 < UNSUPERVISED_F1_TARGET - DEGRADATION_MARGIN:
                alerts.append({
                    "family": family,
                    "metric": "f1",
                    "value": f1,
                    "target": UNSUPERVISED_F1_TARGET,
                    "shortfall": round(UNSUPERVISED_F1_TARGET - f1, 4),
                })
        else:
            acc = metrics.get("accuracy", 1.0)
            if acc < TABULAR_ACCURACY_TARGET - DEGRADATION_MARGIN:
                alerts.append({
                    "family": family,
                    "metric": "accuracy",
                    "value": acc,
                    "target": TABULAR_ACCURACY_TARGET,
                    "shortfall": round(TABULAR_ACCURACY_TARGET - acc, 4),
                })
    global _degradation_alerts
    _degradation_alerts = alerts
    return alerts


def get_degradation_alerts():
    return _degradation_alerts


# --------------------------------------------------------
# 11.2 — Event log (Req 14.3)
# --------------------------------------------------------
def log_event(event_type: str, outcome: str, detail: dict = None):
    """Log a training/recalibration/feedback event (Req 14.3)."""
    database.record_audit_entry(
        category=event_type,
        detail_json=json.dumps({"outcome": outcome, **(detail or {})}),
    )


# --------------------------------------------------------
# Accuracy trend
# --------------------------------------------------------
def get_accuracy_trend(family: str):
    """Return accuracy trend rows for a model family (Req 8.5)."""
    conn = database.get_connection()
    try:
        rows = conn.execute("""
            SELECT at.f1_score, at.measured_at
            FROM accuracy_trend at
            JOIN ml_models m ON at.model_id = m.id
            WHERE m.family = ?
            ORDER BY at.measured_at ASC
        """, (family,)).fetchall()
        return [{"f1_score": r["f1_score"], "measured_at": r["measured_at"]} for r in rows]
    finally:
        conn.close()
