def check_rate_limit(
    scope,
    subject,
    max_attempts=5,
    window_seconds=900,
    block_seconds=900,
):
    if not isinstance(scope, str) or not scope.strip():
        raise ValueError("A valid rate-limit scope is required.")

    _validate_rate_limit_settings(
        max_attempts,
        window_seconds,
        block_seconds,
    )

    subject_hash = _hash_identifier(subject)
    now = _now()
    now_text = now.isoformat()

    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        row = connection.execute("""
            SELECT *
            FROM auth_rate_limits
            WHERE scope = ?
              AND subject_hash = ?
        """, (
            scope,
            subject_hash,
        )).fetchone()

        if row and row["blocked_until"]:
            blocked_until = datetime.fromisoformat(
                row["blocked_until"]
            )

            if blocked_until > now:
                connection.commit()

                return {
                    "allowed": False,
                    "retry_after_seconds": max(
                        1,
                        int((blocked_until - now).total_seconds()),
                    ),
                }

        if row is None:
            connection.execute("""
                INSERT INTO auth_rate_limits (
                    scope,
                    subject_hash,
                    window_started_at,
                    attempts,
                    blocked_until,
                    updated_at
                )
                VALUES (?, ?, ?, 0, NULL, ?)
            """, (
                scope,
                subject_hash,
                now_text,
                now_text,
            ))

            attempts = 0
            window_started = now

        else:
            window_started = datetime.fromisoformat(
                row["window_started_at"]
            )

            attempts = row["attempts"]

            if (now - window_started).total_seconds() >= window_seconds:
                attempts = 0
                window_started = now

                connection.execute("""
                    UPDATE auth_rate_limits
                    SET attempts = 0,
                        window_started_at = ?,
                        blocked_until = NULL,
                        updated_at = ?
                    WHERE scope = ?
                      AND subject_hash = ?
                """, (
                    now_text,
                    now_text,
                    scope,
                    subject_hash,
                ))

        if attempts >= max_attempts:
            blocked_until = now + timedelta(
                seconds=block_seconds
            )

            connection.execute("""
                UPDATE auth_rate_limits
                SET blocked_until = ?,
                    updated_at = ?
                WHERE scope = ?
                  AND subject_hash = ?
            """, (
                blocked_until.isoformat(),
                now_text,
                scope,
                subject_hash,
            ))

            connection.commit()

            return {
                "allowed": False,
                "retry_after_seconds": block_seconds,
            }

        connection.commit()

        return {
            "allowed": True,
            "retry_after_seconds": 0,
        }

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def record_rate_limit_failure(
    scope,
    subject,
    max_attempts=5,
    window_seconds=900,
    block_seconds=900,
):
    if not isinstance(scope, str) or not scope.strip():
        raise ValueError("A valid rate-limit scope is required.")

    _validate_rate_limit_settings(
        max_attempts,
        window_seconds,
        block_seconds,
    )

    subject_hash = _hash_identifier(subject)
    now = _now()
    now_text = now.isoformat()

    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        row = connection.execute("""
            SELECT *
            FROM auth_rate_limits
            WHERE scope = ?
              AND subject_hash = ?
        """, (
            scope,
            subject_hash,
        )).fetchone()

        if row is None:
            attempts = 1
            window_started = now

        else:
            window_started = datetime.fromisoformat(
                row["window_started_at"]
            )

            if (now - window_started).total_seconds() >= window_seconds:
                attempts = 1
                window_started = now
            else:
                attempts = row["attempts"] + 1

        blocked_until = None

        if attempts >= max_attempts:
            blocked_until = (
                now + timedelta(seconds=block_seconds)
            ).isoformat()

        connection.execute("""
            INSERT INTO auth_rate_limits (
                scope,
                subject_hash,
                window_started_at,
                attempts,
                blocked_until,
                updated_at
            )
            VALUES (?, ?, ?, ?, ?, ?)
            ON CONFLICT(scope, subject_hash)
            DO UPDATE SET
                window_started_at = excluded.window_started_at,
                attempts = excluded.attempts,
                blocked_until = excluded.blocked_until,
                updated_at = excluded.updated_at
        """, (
            scope,
            subject_hash,
            window_started.isoformat(),
            attempts,
            blocked_until,
            now_text,
        ))

        connection.commit()

        return {
            "attempts": attempts,
            "blocked": blocked_until is not None,
        }

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()
