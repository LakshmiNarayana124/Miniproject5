class FeedbackRequest(BaseModel):
    name: str
    email: Optional[str] = None
    type: str = "general"
    message: str


@app.post(
    "/feedback",
    tags=["Feedback"],
)
async def submit_feedback(req: FeedbackRequest):
    if not req.name.strip():
        raise HTTPException(
            400,
            "name is required",
        )

    if not req.message.strip():
        raise HTTPException(
            400,
            "message is required",
        )

    obs_service.log_event(
        "feedback",
        "user_feedback",
        {
            "name": req.name.strip(),
            "email": (
                (req.email or "").strip() or None
            ),
            "type": req.type,
            "message": req.message.strip(),
        },
    )

    return {
        "success": True,
        "message": "Feedback received — thank you!",
    }

_leaderboard_state = [
    dict(org)
    for org in _LEADERBOARD_BASE
]

_leaderboard_tick = 0


def _tick_leaderboard():
    global _leaderboard_tick

    _leaderboard_tick += 1

    for org, base in zip(
        _leaderboard_state,
        _LEADERBOARD_BASE,
    ):
        d_score = (
            (_random.random() - 0.5) * 0.4
        )

        d_uptime = (
            (_random.random() - 0.5) * 0.04
        )

        d_mtbf = round(
            (_random.random() - 0.5) * 20
        )

        d_response = (
            (_random.random() - 0.5) * 0.5
        )

        old_score = org["score"]

        new_score = round(
            max(
                base["score"] - 3,
                min(
                    base["score"] + 3,
                    old_score + d_score,
                ),
            )
        )

        new_uptime = round(
            min(
                99.9,
                max(
                    90.0,
                    org["uptime"] + d_uptime,
                ),
            ),
            2,
        )

        new_mtbf = max(
            500,
            org["mtbf"] + d_mtbf,
        )

        new_resp = round(
            max(
                1.0,
                org["responseS"] + d_response
            ),
            1,
        )

        org["score"] = new_score
        org["uptime"] = new_uptime
        org["mtbf"] = new_mtbf
        org["responseS"] = new_resp

        org["trend"] = (
            "up"
            if new_score > old_score
            else "down"
            if new_score < old_score
            else org["trend"]
        )


@app.get(
    "/leaderboard",
    tags=["Leaderboard"],
)
async def get_leaderboard():
    _tick_leaderboard()

    return {
        "tick": _leaderboard_tick,
        "organisations": _leaderboard_state,
        "last_updated": time.time(),
    }
