BASE_DIR = Path(__file__).resolve().parent
DATABASE_DIR = BASE_DIR / "database"
DATABASE_PATH = DATABASE_DIR / "intelliwatch.db"

DATABASE_DIR.mkdir(parents=True, exist_ok=True)

VALID_ROLES = {
    "ADMIN",
    "OPERATOR",
    "MAINTENANCE_ENGINEER",
}

EXPLORER_HISTORY_MAX_ROWS = 500
DEFAULT_RETENTION_DAYS = 365


def get_connection():
    connection = sqlite3.connect(DATABASE_PATH, timeout=30)
    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA foreign_keys = ON")
    connection.execute("PRAGMA busy_timeout = 30000")
    return connection


def current_timestamp():
    return datetime.now(timezone.utc).isoformat()


def _begin_transaction(connection):
    connection.execute("BEGIN IMMEDIATE")


def _insert_audit_entry(
    connection,
    category,
    detail_json,
    actor=None,
    target=None,
    hash_value=None,
):
    """Insert an audit entry using the current transaction."""

    timestamp = current_timestamp()

    if hash_value is None:
        previous = connection.execute("""
            SELECT hash
            FROM audit_entries
            ORDER BY id DESC
            LIMIT 1
        """).fetchone()

        previous_hash = previous["hash"] if previous else ""

        chain_input = "|".join((
            str(previous_hash),
            timestamp,
            str(category),
            str(actor),
            str(target),
            str(detail_json),
        ))

        hash_value = hashlib.sha256(
            chain_input.encode("utf-8")
        ).hexdigest()

    cursor = connection.execute("""
        INSERT INTO audit_entries (
            timestamp,
            category,
            actor,
            target,
            detail_json,
            hash
        )
        VALUES (?, ?, ?, ?, ?, ?)
    """, (
        timestamp,
        category,
        actor,
        str(target) if target is not None else None,
        detail_json,
        hash_value,
    ))

    return cursor.lastrowid


def _audit_details(connection, category, actor, target, details):
    return _insert_audit_entry(
        connection,
        category,
        json.dumps(details or {}, default=str),
        actor=actor,
        target=target,
    )
