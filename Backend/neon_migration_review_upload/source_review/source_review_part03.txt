    username = username.strip()
    full_name = full_name.strip()
    role = role.upper().strip()

    if not username or not full_name or not password_hash:
        raise ValueError("All user details are required.")

    if role not in VALID_ROLES:
        raise ValueError("Invalid user role.")

    connection = get_connection()

    try:
        cursor = connection.execute("""
            INSERT INTO users (
                username, full_name, password_hash,
                role, created_at
            )
            VALUES (?, ?, ?, ?, ?)
        """, (
            username,
            full_name,
            password_hash,
            role,
            current_timestamp(),
        ))

        connection.commit()
        return cursor.lastrowid

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def set_user_active(user_id, is_active, actor=None):
    """
    Activate or deactivate an account transactionally.

    The last active Admin cannot be deactivated.
    """

    connection = get_connection()

    try:
        _begin_transaction(connection)

        target = connection.execute("""
            SELECT id, username, role, is_active, is_deleted
            FROM users
            WHERE id = ?
        """, (user_id,)).fetchone()

        if not target or target["is_deleted"]:
            connection.rollback()
            return False

        active = int(bool(is_active))

        if (
            target["role"] == "ADMIN"
            and target["is_active"]
            and not active
        ):
            count = connection.execute("""
                SELECT COUNT(*) AS total
                FROM users
                WHERE role = 'ADMIN'
                  AND is_active = 1
                  AND is_deleted = 0
            """).fetchone()["total"]

            if count <= 1:
                raise ValueError(
                    "The last active administrator cannot be deactivated."
                )

        cursor = connection.execute("""
            UPDATE users
            SET is_active = ?
            WHERE id = ? AND is_deleted = 0
        """, (active, user_id))

        if cursor.rowcount:
            _audit_details(
                connection,
                "USER_UNLOCKED" if active else "USER_LOCKED",
                actor,
                user_id,
                {
                    "username": target["username"],
                    "role": target["role"],
                    "is_active": bool(active),
                },
            )

        connection.commit()
        return cursor.rowcount > 0

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def delete_user(user_id: int, deleted_by=None, reason=None) -> bool:
    """
    Temporarily delete an Operator or Maintenance Engineer.

    The user record, permissions and related history are preserved.
    """

    connection = get_connection()

    try:
        _begin_transaction(connection)

        target = connection.execute("""
            SELECT *
            FROM users
            WHERE id = ?
        """, (user_id,)).fetchone()

        if (
            not target
            or target["is_deleted"]
            or target["role"] == "ADMIN"
        ):
            connection.rollback()
            return False

        connection.execute("""
            UPDATE users
            SET is_deleted = 1,
                is_active = 0,
                deleted_at = ?,
                deleted_by = ?,
                deletion_reason = ?,
                deleted_was_active = is_active
            WHERE id = ?
        """, (
            current_timestamp(),
            str(deleted_by) if deleted_by is not None else None,
            reason,
            user_id,
        ))

        _audit_details(
            connection,
            "USER_TEMPORARILY_DELETED",
            deleted_by,
            user_id,
            {
                "username": target["username"],
                "role": target["role"],
                "reason": reason,
                "previously_active": bool(target["is_active"]),
            },
        )

        connection.commit()
        return True

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def restore_user(user_id: int, restored_by=None) -> bool:
    """Restore a temporarily deleted account and its previous active state."""

    connection = get_connection()

    try:
        _begin_transaction(connection)

        target = connection.execute("""
            SELECT *
            FROM users
            WHERE id = ?
        """, (user_id,)).fetchone()

        if not target or not target["is_deleted"]:
            connection.rollback()
            return False

        connection.execute("""
            UPDATE users
            SET is_deleted = 0,
                is_active = deleted_was_active,
                deleted_at = NULL,
                deleted_by = NULL,
                deletion_reason = NULL,
                deleted_was_active = 1
            WHERE id = ?
        """, (user_id,))

        _audit_details(
            connection,
            "USER_RESTORED",
            restored_by,
            user_id,
            {
                "username": target["username"],
                "previous_role": target["role"],
            },
        )

        connection.commit()
        return True

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def permanently_delete_user(user_id: int, deleted_by=None) -> bool:
    """
    Permanently delete a previously soft-deleted non-Admin account.

    The permanent-deletion audit entry is inserted before the account
    is removed. Audit history is intentionally independent of users.
    """

    connection = get_connection()

    try:
        _begin_transaction(connection)

        target = connection.execute("""
            SELECT *
            FROM users
            WHERE id = ?
        """, (user_id,)).fetchone()

        if (
            not target
            or not target["is_deleted"]
            or target["role"] == "ADMIN"
        ):
            connection.rollback()
            return False

        _audit_details(
            connection,
            "USER_PERMANENTLY_DELETED",
            deleted_by,
            user_id,
            {
                "username": target["username"],
                "full_name": target["full_name"],
                "role": target["role"],
                "deleted_at": target["deleted_at"],
                "reason": target["deletion_reason"],
                "permanently_deleted_at": current_timestamp(),
            },
        )

        connection.execute(
            "DELETE FROM users WHERE id = ?",
            (user_id,),
        )

        connection.commit()
        return True

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()
