def cleanup_old_records(retention_days=None):
    """
    Retain sensor and anomaly data for the configured period.

    Audit entries are deliberately excluded to preserve security history.
    """

    days = (
        DEFAULT_RETENTION_DAYS
        if retention_days is None
        else max(1, int(retention_days))
    )

    cutoff = (
        datetime.now(timezone.utc) - timedelta(days=days)
    ).isoformat()

    deleted = {}
    connection = get_connection()

    try:
        for table in ("sensor_readings", "anomaly_events"):
            cursor = connection.execute(
                f"DELETE FROM {table} WHERE timestamp < ?",
                (cutoff,),
            )

            deleted[table] = cursor.rowcount

        deleted["audit_entries"] = 0

        connection.commit()
        return deleted

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def set_user_permission(
    user_id,
    permission_key,
    is_granted,
    updated_by=None,
):
    if not isinstance(permission_key, str) or not permission_key.strip():
        raise ValueError("A valid permission key is required.")

    if type(is_granted) is not bool:
        raise ValueError("is_granted must be a boolean.")

    connection = get_connection()

    try:
        connection.execute("""
            INSERT INTO user_permissions (
                user_id, permission_key, is_granted,
                updated_by, updated_at
            )
            VALUES (?, ?, ?, ?, ?)
            ON CONFLICT(user_id, permission_key)
            DO UPDATE SET
                is_granted = excluded.is_granted,
                updated_by = excluded.updated_by,
                updated_at = excluded.updated_at
        """, (
            user_id,
            permission_key.strip(),
            int(is_granted),
            updated_by,
            current_timestamp(),
        ))

        connection.commit()
        return True

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def clear_user_permission_override(user_id, permission_key):
    connection = get_connection()

    try:
        cursor = connection.execute("""
            DELETE FROM user_permissions
            WHERE user_id = ?
              AND permission_key = ?
        """, (
            user_id,
            permission_key,
        ))

        connection.commit()
        return cursor.rowcount > 0

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def _now():
    return datetime.now(timezone.utc).isoformat()


def _audit_event(category, actor=None, target=None, details=None):
    record_audit_entry(
        category=category,
        actor=actor,
        target=str(target) if target is not None else None,
        detail_json=json.dumps(details or {}, default=str),
    )


def _begin_transaction():
    connection = get_connection()
    connection.execute("BEGIN IMMEDIATE")
    return connection


def _audit_transaction(connection, category, actor, target, details):
    timestamp = _now()
    detail_json = json.dumps(details or {}, default=str)

    previous = connection.execute("""
        SELECT hash
        FROM audit_entries
        ORDER BY id DESC
        LIMIT 1
    """).fetchone()

    previous_hash = previous["hash"] if previous else ""

    chain_input = "|".join((
        str(previous_hash),
        timestamp,
        str(category),
        str(actor),
        str(target),
        detail_json,
    ))

    entry_hash = hashlib.sha256(
        chain_input.encode("utf-8")
    ).hexdigest()

    cursor = connection.execute("""
        INSERT INTO audit_entries (
            timestamp, category, actor, target, detail_json, hash
        )
        VALUES (?, ?, ?, ?, ?, ?)
    """, (
        timestamp,
        category,
        actor,
        str(target) if target is not None else None,
        detail_json,
        entry_hash,
    ))

    return cursor.lastrowid


def _admin_in_transaction(connection, admin_id):
    admin = connection.execute("""
        SELECT id, username, role, is_active, is_deleted
        FROM users
        WHERE id = ?
    """, (admin_id,)).fetchone()

    if (
        not admin
        or admin["role"] != "ADMIN"
        or not admin["is_active"]
        or admin["is_deleted"]
    ):
        raise ValueError("An active administrator is required.")

    return dict(admin)
