logger = logging.getLogger(__name__)


def _now():
    return datetime.now(timezone.utc).isoformat()


class LearningSystem:
    """Accept operator labels and recalibrate/promote models (Req 8.1-8.5)."""

    def __init__(self, registry=None):
        self.registry = registry or _mr.registry

    # --------------------------------------------------------
    # 10.1 — submit_label  (Req 8.1, 8.2)
    # --------------------------------------------------------
    def submit_label(self, event_id, label, feature_vector, model_version, verified_by):
        """Store an operator-verified label with its feature vector (Req 8.1, 8.2)."""
        fv_json = json.dumps(feature_vector, default=str) if not isinstance(feature_vector, str) else feature_vector
        row_id = database.save_verified_label(
            event_id=event_id,
            label=label,
            feature_vector_json=fv_json,
            model_version=model_version,
            verified_by=verified_by,
        )
        # Log for event-log completeness (Req 14.3)
        database.record_audit_entry(
            category="feedback_ingestion",
            detail_json=json.dumps({"event_id": event_id, "label": label, "model_version": model_version}),
            actor=verified_by,
        )
        return row_id

    # --------------------------------------------------------
    # 10.3 — recalibrate  (Req 8.3-8.5, 13.2)
    # --------------------------------------------------------
    def recalibrate(self, family="tabular"):
        """Train a candidate from accumulated labels and promote if better (Req 8.3-8.5)."""
        # Load accumulated verified labels
        labels_rows = self._load_labels()
        if not labels_rows:
            logger.info("No verified labels available for recalibration.")
            return {"promoted": False, "reason": "no labels"}

        # Build feature matrix from stored vectors
        try:
            X, y = self._build_dataset(labels_rows)
        except Exception as exc:
            logger.warning("Could not build dataset for recalibration: %s", exc)
            return {"promoted": False, "reason": str(exc)}

        # Train a candidate (Req 8.3)
        try:
            from training_pipeline import train_tabular, validate_and_report, persist_model, _check_promotion_gate
            import pandas as pd
            candidate = train_tabular(X, y, version=_now()[:19].replace(":", "").replace("-", "").replace("T", "_"))
        except Exception as exc:
            logger.warning("Recalibration training failed: %s", exc)
            return {"promoted": False, "reason": str(exc)}

        # Validate candidate
        try:
            report = validate_and_report({"tabular": candidate}, X, y)
            metrics = report.get("tabular", {})
        except Exception as exc:
            metrics = {"f1": 0.0, "accuracy": 0.0, "meets_target": False}

        candidate_f1 = metrics.get("f1", 0.0)

        # Get active model's F1 for comparison (Req 8.4)
        active_model = self.registry.active_model(family)
        active_f1 = 0.0
        if active_model is not None:
            active_manifest = self.registry.get_manifest(family, getattr(active_model, "version", ""))
            if active_manifest:
                active_f1 = active_manifest.get("metrics", {}).get("f1", 0.0)

        promoted = False
        gate_passed = _check_promotion_gate("tabular", metrics)

        if gate_passed and candidate_f1 > active_f1:
            # Promote (Req 8.4)
            try:
                import os
                out_dir = os.path.join(os.path.dirname(__file__), "models")
                persist_model(candidate, family, out_dir, dataset_hash="recalibration", metrics=metrics)
                self.registry.register(candidate, family=family, version=candidate.version, metrics=metrics)
                self.registry.set_active(family, candidate.version)
                promoted = True
            except Exception as exc:
                logger.warning("Promotion failed: %s", exc)

        # Append accuracy trend (Req 8.5)
        try:
            self._append_trend(family, candidate.version, candidate_f1)
        except Exception as exc:
            logger.warning("Could not append accuracy trend: %s", exc)

        # Audit entry (Req 13.2)
        action = "model_promotion" if promoted else "recalibration"
        database.record_audit_entry(
            category=action,
            detail_json=json.dumps({
                "family": family,
                "version": getattr(candidate, "version", "?"),
                "metrics": metrics,
                "promoted": promoted,
                "candidate_f1": candidate_f1,
                "active_f1": active_f1,
            }),
        )

        return {"promoted": promoted, "metrics": metrics, "candidate_f1": candidate_f1}

    # --------------------------------------------------------
    # Helpers
    # --------------------------------------------------------
    def _load_labels(self):
        conn = database.get_connection()
        try:
            rows = conn.execute(
                "SELECT * FROM verified_labels ORDER BY verified_at ASC"
            ).fetchall()
            return [dict(r) for r in rows]
        finally:
            conn.close()

    def _build_dataset(self, rows):
        """Return (X, y) from stored verified_labels rows."""
        try:
            import pandas as pd
            records = []
            labels = []
            for row in rows:
                fv = json.loads(row["feature_vector_json"])
                if isinstance(fv, dict):
                    records.append(fv)
                elif isinstance(fv, list):
                    records.append({f"f{i}": v for i, v in enumerate(fv)})
                labels.append(row["label"])
            X = pd.DataFrame(records).fillna(0.0)
            return X, labels
        except Exception:
            raise

    def _append_trend(self, family, version, f1):
        conn = database.get_connection()
        try:
            row = conn.execute(
                "SELECT id FROM ml_models WHERE family=? AND version=?",
                (family, version)
            ).fetchone()
            if row:
                database.append_accuracy_trend(row["id"], f1)
        finally:
            conn.close()

Version: 1.0

- Thread-safe registration, activation, and lookup keyed by
  family and version.
- Exactly one active model version per family (ensemble by default).
- SQLite persistence via the ``ml_models`` table.
- On-disk manifests under ``Backend/models/{family}/{version}/manifest.json``.

Dependencies:
- Python standard library.
- Existing ``database.py`` (``get_connection``).

import database

# =====================================================

BASE_DIR = Path(__file__).resolve().parent
MODELS_DIR = BASE_DIR / "models"

SCHEMA_VERSION = "1.0"

DEFAULT_FAMILY = "ensemble"


def _current_timestamp() -> str:
    """Return a timezone-aware UTC timestamp."""

    return datetime.now(timezone.utc).isoformat()


# =====================================================
# MODEL REGISTRY
# =====================================================

class ModelRegistry:
    """
    Thread-safe in-memory + on-disk catalog of trained models.

    Each model is keyed by ``(family, version)``. Only one version
    per family may be active at a time. Registering a model writes a
    ``manifest.json`` to disk and upserts a row in the ``ml_models``
    table, then keeps the live model object in memory so the in-process
    inference path can use it without IPC overhead.
    """

    def __init__(self, models_dir: Optional[Path] = None) -> None:
        self._models_dir = Path(models_dir or MODELS_DIR)
        self._models_dir.mkdir(parents=True, exist_ok=True)

        self._lock = threading.Lock()

        # (family, version) -> catalog entry holding the model object
        # and its manifest metadata.
        self._catalog: dict[tuple[str, str], dict[str, Any]] = {}

        # family -> currently active version.
        self._active: dict[str, str] = {}

    # -------------------------------------------------
    # REGISTER
    # -------------------------------------------------

    def register(
        self,
        model: Any,
        family: str,
        version: str,
        metrics: Optional[dict[str, Any]] = None,
        dataset_hash: Optional[str] = None,
        artifact_path: Optional[str] = None
    ) -> dict[str, Any]:
        """
        Register a trained model in the in-memory catalog, write its
        manifest to disk, and upsert a row in the ``ml_models`` table.

        The model is registered inactive; call ``set_active`` to make it
        the live model for its family.

        Returns the manifest dictionary that was persisted.
        """

        family = str(family).strip()
        version = str(version).strip()

        if not family or not version:
            raise ValueError("family and version are required.")

        dataset_hash = dataset_hash or ""
        metrics = metrics or {}
        model_name = type(model).__name__

        # Resolve the on-disk artifact directory.
        artifact_dir = self._models_dir / family / version
        artifact_dir.mkdir(parents=True, exist_ok=True)

        artifact_path = str(artifact_path) if artifact_path else str(artifact_dir)

        training_timestamp = _current_timestamp()

        manifest = {
            "family": family,
            "model_name": model_name,
            "version": version,
            "training_timestamp": training_timestamp,
            "dataset_hash": dataset_hash,
            "artifact_path": artifact_path,
            "metrics": metrics,
            "schema_version": SCHEMA_VERSION
        }

        # 1. Write the manifest to disk.
        manifest_path = artifact_dir / "manifest.json"
        manifest_path.write_text(
            json.dumps(manifest, indent=2, default=str),
            encoding="utf-8"
        )

        # 2. Persist to the database.
        self._upsert_model_row(
            family=family,
            model_name=model_name,
            version=version,
            training_timestamp=training_timestamp,
            dataset_hash=dataset_hash,
            artifact_path=artifact_path,
            metrics=metrics
        )

        # 3. Store in the in-memory catalog (thread-safe).
        entry = {
            "model": model,
            "manifest": manifest,
            "is_active": self._active.get(family) == version
        }

        with self._lock:
            self._catalog[(family, version)] = entry

        return manifest

    # -------------------------------------------------
    # SET ACTIVE
    # -------------------------------------------------

    def set_active(self, family: str, version: str) -> bool:
        """
        Mark ``version`` as the active model for ``family``.

        Exactly one version per family is active at a time, so every
        other version in the family is deactivated first. The change is
        persisted to the ``ml_models`` table and mirrored in the
        in-memory catalog.

        Returns ``True`` if a matching registered model was found,
        otherwise ``False``.
        """

        family = str(family).strip()
        version = str(version).strip()

        if not family or not version:
            raise ValueError("family and version are required.")

        with self._lock:
            # The model must be registered (in memory or database).
            if (family, version) not in self._catalog:
                if not self._model_row_exists(family, version):
                    return False

            # Deactivate every other version in this family.
            self._deactivate_family_rows(family)

            # Activate the selected version.
            self._activate_model_row(family, version)

            # Mirror the active state in memory.
            self._active[family] = version

            for key, entry in self._catalog.items():
                if key[0] == family:
                    entry["is_active"] = (key[1] == version)

        return True

    # -------------------------------------------------
    # ACTIVE MODEL
    # -------------------------------------------------

    def active_model(self, family: str = DEFAULT_FAMILY) -> Any:
        """
        Return the currently active model object for ``family``,
        or ``None`` when no model is active or the model object has not
        been loaded into memory.
        """

        family = str(family).strip()

        with self._lock:
            version = self._active.get(family)

            if version is None:
                version = self._active_version_from_db(family)

                if version is None:
                    return None

                self._active[family] = version

            entry = self._catalog.get((family, version))

        if entry is None:
            return None

        return entry["model"]
