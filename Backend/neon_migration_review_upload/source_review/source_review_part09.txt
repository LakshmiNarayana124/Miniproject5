async def list_admin_invites(current_admin=Depends(admin_access)):
    connection = get_connection()

    try:
        rows = connection.execute("""
            SELECT
                id,
                created_by_user_id,
                created_at,
                expires_at
            FROM auth_security_tokens
            WHERE token_type = 'ADMIN_INVITE'
              AND consumed_at IS NULL
              AND revoked_at IS NULL
              AND expires_at > ?
            ORDER BY created_at DESC
        """, (_now(),)).fetchall()

        active = []

        for row in rows:
            creator = get_user_by_id(row["created_by_user_id"])
            active.append({
                "created_by": (
                    creator["username"] if creator else "Unavailable"
                ),
                "created_at": row["created_at"],
                "expires_at": row["expires_at"],
            })

        return {
            "pending_invites": active,
            "count": len(active),
        }

    finally:
        connection.close()

@auth_router.post("/register-admin")
async def register_admin(
    request: AdminRegisterRequest,
    http_request: Request,
):
    rate_key = _enforce_rate_limit(
        "register_admin",
        http_request,
        request.username,
        max_attempts=3,
        window_seconds=900,
    )

    try:
        new_hash = hash_password(request.password)
    except ValueError as error:
        raise HTTPException(status_code=400, detail=str(error))

    # consume_token must atomically mark the invitation as consumed.
    try:
        entry = consume_token(
            request.invite_token.strip(),
            "ADMIN_INVITE",
        )
    except Exception:
        entry = None

    if not entry:
        _record_failure(rate_key)
        raise HTTPException(
            status_code=403,
            detail="Invalid, expired or already used Admin invitation.",
        )

    inviter_id = entry.get("created_by_user_id")

    if not inviter_id:
        _record_failure(rate_key)
        raise HTTPException(
            status_code=403,
            detail="The invitation creator is no longer valid.",
        )

    inviter = get_user_by_id(inviter_id)

    if (
        not inviter
        or inviter["role"] != "ADMIN"
        or not inviter["is_active"]
        or inviter.get("is_deleted", 0)
    ):
        _record_failure(rate_key)
        raise HTTPException(
            status_code=403,
            detail="The invitation creator is no longer an active Admin.",
        )

    try:
        user_id = create_user(
            username=request.username,
            full_name=request.full_name,
            password_hash=new_hash,
            role="ADMIN",
        )

        _clear_failures(rate_key)

        _audit_event(
            "ADMIN_REGISTERED_BY_INVITATION",
            actor=inviter["username"],
            target=user_id,
            details={"username": request.username},
        )

    except sqlite3.IntegrityError:
        _record_failure(rate_key)
        raise HTTPException(status_code=409, detail="Username already exists.")

    except ValueError as error:
        _record_failure(rate_key)
        raise HTTPException(status_code=400, detail=str(error))

    return {
        "success": True,
        "user_id": user_id,
        "username": request.username,
        "role": "ADMIN",
        "invited_by": inviter["username"],
        "message": "Co-Admin account created successfully.",
    }

import hashlib
import sqlite3
from datetime import datetime, timezone, timedelta
from pathlib import Path


BASE_DIR = Path(__file__).resolve().parent
DATABASE_PATH = BASE_DIR / "database" / "intelliwatch.db"

TOKEN_TYPES = {"PASSWORD_RESET", "ADMIN_INVITE"}


# =========================================================
# SHARED HELPERS
# =========================================================

def _now():
    return datetime.now(timezone.utc)


def _timestamp(value=None):
    if value is None:
        value = _now()

    if not isinstance(value, datetime):
        raise ValueError("A valid datetime is required.")

    if value.tzinfo is None or value.utcoffset() is None:
        raise ValueError("Datetime values must include a timezone.")

    return value.astimezone(timezone.utc).isoformat()


def _hash_identifier(value):
    if not isinstance(value, str) or not value:
        raise ValueError("A valid identifier is required.")

    return hashlib.sha256(
        value.encode("utf-8")
    ).hexdigest()


def _validate_token_type(token_type):
    if token_type not in TOKEN_TYPES:
        raise ValueError("Invalid security token type.")


def _validate_rate_limit_settings(
    max_attempts,
    window_seconds,
    block_seconds,
):
    for value in (max_attempts, window_seconds, block_seconds):
        if isinstance(value, bool) or not isinstance(value, int) or value < 1:
            raise ValueError("Rate-limit settings must be positive integers.")


def _connect():
    connection = sqlite3.connect(
        DATABASE_PATH,
        timeout=30,
        isolation_level=None,
    )

    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA foreign_keys = ON")
    connection.execute("PRAGMA busy_timeout = 30000")

    return connection

def initialize_security_storage():
    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        connection.execute("""
            CREATE TABLE IF NOT EXISTS auth_security_tokens (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                token_hash TEXT NOT NULL UNIQUE,
                token_type TEXT NOT NULL
                    CHECK (
                        token_type IN (
                            'PASSWORD_RESET',
                            'ADMIN_INVITE'
                        )
                    ),
                user_id INTEGER,
                created_by_user_id INTEGER,
                created_at TEXT NOT NULL,
                expires_at TEXT NOT NULL,
                consumed_at TEXT,
                revoked_at TEXT,
                metadata_json TEXT,
                FOREIGN KEY (user_id)
                    REFERENCES users(id) ON DELETE CASCADE,
                FOREIGN KEY (created_by_user_id)
                    REFERENCES users(id) ON DELETE SET NULL
            )
        """)

        connection.execute("""
            CREATE TABLE IF NOT EXISTS auth_rate_limits (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                scope TEXT NOT NULL,
                subject_hash TEXT NOT NULL,
                window_started_at TEXT NOT NULL,
                attempts INTEGER NOT NULL DEFAULT 0
                    CHECK (attempts >= 0),
                blocked_until TEXT,
                updated_at TEXT NOT NULL,
                UNIQUE (scope, subject_hash)
            )
        """)

        columns = {
            row["name"]
            for row in connection.execute(
                "PRAGMA table_info(users)"
            ).fetchall()
        }

        if "session_version" not in columns:
            connection.execute("""
                ALTER TABLE users
                ADD COLUMN session_version INTEGER NOT NULL DEFAULT 0
                    CHECK (session_version >= 0)
            """)

        connection.execute("""
            CREATE INDEX IF NOT EXISTS idx_auth_tokens_type_expiry
            ON auth_security_tokens(token_type, expires_at)
        """)

        connection.execute("""
            CREATE INDEX IF NOT EXISTS idx_auth_tokens_user
            ON auth_security_tokens(user_id)
        """)

        connection.execute("""
            CREATE INDEX IF NOT EXISTS idx_auth_rate_blocked
            ON auth_rate_limits(blocked_until)
        """)

        connection.commit()

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()

def store_token(
    token,
    token_type,
    expires_at,
    user_id=None,
    created_by_user_id=None,
    metadata_json=None,
):
    _validate_token_type(token_type)

    token_hash = _hash_identifier(token)
    expiry = _timestamp(expires_at)

    if metadata_json is not None and not isinstance(metadata_json, str):
        raise ValueError("metadata_json must be a string or None.")

    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        cursor = connection.execute("""
            INSERT INTO auth_security_tokens (
                token_hash,
                token_type,
                user_id,
                created_by_user_id,
                created_at,
                expires_at,
                metadata_json
            )
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (
            token_hash,
            token_type,
            user_id,
            created_by_user_id,
            _timestamp(),
            expiry,
            metadata_json,
        ))

        connection.commit()
        return cursor.lastrowid

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def consume_token(token, token_type):
    _validate_token_type(token_type)

    token_hash = _hash_identifier(token)
    now = _timestamp()

    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        row = connection.execute("""
            SELECT *
            FROM auth_security_tokens
            WHERE token_hash = ?
              AND token_type = ?
              AND consumed_at IS NULL
              AND revoked_at IS NULL
              AND expires_at > ?
        """, (
            token_hash,
            token_type,
            now,
        )).fetchone()

        if row is None:
            connection.rollback()
            return None

        cursor = connection.execute("""
            UPDATE auth_security_tokens
            SET consumed_at = ?
            WHERE id = ?
              AND consumed_at IS NULL
              AND revoked_at IS NULL
              AND expires_at > ?
        """, (
            now,
            row["id"],
            now,
        ))

        if cursor.rowcount != 1:
            connection.rollback()
            return None

        connection.commit()
        return dict(row)

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def revoke_token(token, token_type):
    _validate_token_type(token_type)

    token_hash = _hash_identifier(token)

    connection = _connect()

    try:
        connection.execute("BEGIN IMMEDIATE")

        cursor = connection.execute("""
            UPDATE auth_security_tokens
            SET revoked_at = ?
            WHERE token_hash = ?
              AND token_type = ?
              AND consumed_at IS NULL
              AND revoked_at IS NULL
        """, (
            _timestamp(),
            token_hash,
            token_type,
        ))

        connection.commit()
        return cursor.rowcount == 1

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()
