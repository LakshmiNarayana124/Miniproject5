NAME: account_recovery_requests
TABLE: account_recovery_requests
SQL: CREATE TABLE account_recovery_requests (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                recovery_reason TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'PENDING'
                    CHECK (status IN ('PENDING', 'APPROVED', 'REJECTED')),
                reviewed_by INTEGER,
                review_remarks TEXT,
                requested_at TEXT NOT NULL,
                reviewed_at TEXT,
                FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
                FOREIGN KEY (reviewed_by) REFERENCES users(id) ON DELETE SET NULL
            )
FOREIGN KEY: (0, 0, 'users', 'reviewed_by', 'id', 'NO ACTION', 'SET NULL', 'NONE')
FOREIGN KEY: (1, 0, 'users', 'user_id', 'id', 'NO ACTION', 'CASCADE', 'NONE')
