def _audit_transaction(connection, category, actor, target, details):
    timestamp = _now()
    detail_json = json.dumps(details or {}, default=str)

    previous = connection.execute("""
        SELECT hash
        FROM audit_entries
        ORDER BY id DESC
        LIMIT 1
    """).fetchone()

    previous_hash = previous["hash"] if previous else ""

    chain_input = "|".join((
        str(previous_hash),
        timestamp,
        str(category),
        str(actor),
        str(target),
        detail_json,
    ))

    entry_hash = hashlib.sha256(
        chain_input.encode("utf-8")
    ).hexdigest()

    cursor = connection.execute("""
        INSERT INTO audit_entries (
            timestamp, category, actor, target, detail_json, hash
        )
        VALUES (?, ?, ?, ?, ?, ?)
    """, (
        timestamp,
        category,
        actor,
        str(target) if target is not None else None,
        detail_json,
        entry_hash,
    ))

    return cursor.lastrowid

def _client_ip(request):
    """
    Uses the direct connection address.

    Do not trust X-Forwarded-For unless a trusted reverse proxy
    is configured to overwrite and validate it.
    """
    if not request or not request.client:
        return "unknown"

    host = request.client.host or "unknown"

    try:
        return str(ipaddress.ip_address(host))
    except ValueError:
        return "unknown"

def _rate_limit_key(action, request, subject=""):
    """
    Creates a rate-limit scope and a privacy-preserving subject.

    The subject includes the action, client IP and supplied account
    identifier. Only its SHA-256 hash is passed to the storage layer.
    """
    raw = (
        f"{action}|{_client_ip(request)}|"
        f"{str(subject).strip().lower()}"
    )

    subject_hash = hashlib.sha256(
        raw.encode("utf-8")
    ).hexdigest()

    return f"{action}:{subject_hash}"


def _split_rate_limit_key(key):
    """
    Converts the internal combined key into the two arguments
    required by auth_security_store.py: scope and subject.
    """
    scope, separator, subject = str(key).partition(":")

    if not separator or not scope or not subject:
        raise ValueError("Invalid rate-limit key.")

    return scope, subject


def _enforce_rate_limit(
    action,
    request,
    subject="",
    max_attempts=5,
    window_seconds=300,
):

def hash_password(password: str) -> str:
    if not isinstance(password, str):
        raise ValueError("Password must be a string.")

    if not 12 <= len(password) <= 128:
        raise ValueError(
            "Password must contain between 12 and 128 characters."
        )

    return password_hasher.hash(password)


def verify_password(plain_password, password_hash):
    if not isinstance(plain_password, str) or not isinstance(
        password_hash, str
    ):
        return False

    try:
        return password_hasher.verify(password_hash, plain_password)
    except (
        VerifyMismatchError,
        VerificationError,
        InvalidHashError,
        TypeError,
        ValueError,
    ):
        return False

def get_effective_permissions(user_id, role, restricted_session=False):
    role = str(role).upper()

    permissions = set(ROLE_DEFAULT_PERMISSIONS.get(role, set()))

    if restricted_session:
        permissions.difference_update(ADMIN_ONLY_PERMISSIONS)
        return sorted(permissions)

    overrides = get_user_permission_overrides(user_id)

    for key, granted in overrides.items():
        if key not in ALL_PERMISSION_KEYS:
            continue

        if role != "ADMIN" and key in ADMIN_ONLY_PERMISSIONS:
            continue

        if granted:
            permissions.add(key)
        else:
            permissions.discard(key)

def get_all_users(include_deleted=False):
    connection = get_connection()

    try:
        query = """
            SELECT
                id, username, full_name, role,
                is_active, created_at, last_login,
                is_deleted, deleted_at
            FROM users
        """

        if not include_deleted:
            query += " WHERE is_deleted = 0"

        query += " ORDER BY id"

        rows = connection.execute(query).fetchall()
        return [dict(row) for row in rows]

    finally:
        connection.close()

        raise

    finally:
        connection.close()

function isBrowserOnline() {

function getLoginErrorMessage(data, fallback) {
  if (typeof data?.detail === 'string' && data.detail.trim()) {
    return data.detail
  }

  if (Array.isArray(data?.detail)) {
    const messages = data.detail
      .map(item => item?.msg)
      .filter(message => typeof message === 'string')

    if (messages.length > 0) {
      return messages.join('; ')
    }
  }

  if (
    data?.detail &&
    typeof data.detail === 'object' &&
    typeof data.detail.message === 'string'
  ) {
    return data.detail.message
  }

  if (typeof data?.message === 'string' && data.message.trim()) {
    return data.message
  }

  return fallback
}

function saveSession(token, user) {
  if (
    typeof token !== 'string' ||
    !token.trim() ||
    !user ||
    typeof user !== 'object'
  ) {
    throw new Error('Invalid authentication session.')
  }

  const serializedUser = JSON.stringify(user)

  try {
    sessionStorage.setItem('iw_token', token)
    sessionStorage.setItem('iw_user', serializedUser)
  } catch (error) {
    // Do not leave an incomplete session behind.
    try {
      sessionStorage.removeItem('iw_token')
      sessionStorage.removeItem('iw_user')
    } catch (cleanupError) {
      console.warn(
        '[IntelliWatch] Session cleanup failed:',
        cleanupError
      )
    }

    throw error
  }
}

function clearSavedSession() {
  try {
    sessionStorage.removeItem('iw_token')
    sessionStorage.removeItem('iw_user')
  } catch (error) {
    console.warn(
      '[IntelliWatch] Unable to clear stored session:',
      error
    )
  }
}

async function fetchWithTimeout(url, options = {}) {
  const controller = new AbortController()

  const timeout = setTimeout(
    () => controller.abort(),
    AVAILABILITY_TIMEOUT_MS
  )

  try {
    return await fetch(url, {
      ...options,
      signal: controller.signal,
    })
  } finally {
    clearTimeout(timeout)
  }
}

async function checkInternetAvailability() {
  if (!isBrowserOnline()) return false

  try {
    await fetchWithTimeout(INTERNET_CHECK_URL, {
      method: 'GET',
      mode: 'no-cors',
      cache: 'no-store',
    })

    return true
  } catch {
    return false
  }
}

      return {
        available: false,
        registrationEnabled: false,
      }
    }

    const data = await response.json()

    return {
      available: true,
      registrationEnabled: !!data.registration_enabled,
    }
  } catch {
    return {
      available: false,
      registrationEnabled: false,
    }
  }
}

async function detectAvailability() {
  const [internetAvailable, backend] = await Promise.all([
    checkInternetAvailability(),
    checkBackendAvailability(),
  ])

  return {
    checked: true,
    internetAvailable,
    backendAvailable: backend.available,
    onlineMode: backend.available,
    registrationEnabled: backend.registrationEnabled,
  }
}

// 4. PASSWORD STRENGTH
// ─────────────────────────────────────────────────────────────────────────────

function PasswordStrength({ password }) {
  if (!password) return null

  const score = [
    password.length >= 12,
    /[A-Z]/.test(password),
    /[a-z]/.test(password),
    /\d/.test(password),
    /[^A-Za-z0-9]/.test(password),
  ].filter(Boolean).length

  const labels = ['', 'Very Weak', 'Weak', 'Fair', 'Strong', 'Very Strong']
  const colors = ['', '#ef4444', '#f59e0b', '#eab308', '#22c55e', '#10b981']

  return (
    <div className="pw-strength">
      <div className="pw-bars">
        {[1, 2, 3, 4, 5].map(i => (
          <div
            key={i}
            className="pw-bar"
            style={{
              background: i <= score ? colors[score] : '#252837',
            }}
          />
        ))}
      </div>

      <span className="pw-label" style={{ color: colors[score] }}>
        {labels[score]}
      </span>
    </div>
  )
}

function ForgotModal({ onClose }) {
  const [step, setStep] = useState('request')
  const [username, setUsername] = useState('')
  const [token, setToken] = useState('')
  const [newPw, setNewPw] = useState('')
  const [confirmPw, setConfirmPw] = useState('')
  const [showPw, setShowPw] = useState(false)
  const [error, setError] = useState('')
  const [info, setInfo] = useState('')
  const [loading, setLoading] = useState(false)
  const [done, setDone] = useState(false)

  async function handleRequest(e) {
    e.preventDefault()
    setError('')

    if (!username.trim()) {
      setError('Please enter your username.')
      return
    }

    setLoading(true)

    try {
      const r = await fetch(`${API}/auth/forgot-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username: username.trim() }),
      })

      const d = await r.json().catch(() => ({}))

      if (!r.ok) {
        setError(
          getLoginErrorMessage(d, 'Unable to process your request.')
        )
        return
      }

      if (d.reset_token) {
        setInfo(
          `Reset Token: ${d.reset_token}\n\nKeep this token secure and use it to reset your password.`
        )
      } else {
        setInfo(d.note || d.message || '')
      }

      setStep('reset')
    } catch (error) {
      console.error(
        '[IntelliWatch] Password-reset request failed:',
        error
      )

      setError('Unable to connect to the server. Please try again.')
    } finally {
      setLoading(false)
    }
  }

  async function handleReset(e) {
    e.preventDefault()
    setError('')

    if (!token.trim()) {
      setError('Please enter the reset token.')
      return
    }

    if (newPw.length < 12) {
      setError('Password must be at least 12 characters long.')
      return
    }

    if (newPw !== confirmPw) {
      setError('Passwords do not match.')
      return
    }

    setLoading(true)

    try {
      const r = await fetch(`${API}/auth/reset-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          token: token.trim(),
          new_password: newPw,
        }),
      })

      const d = await r.json().catch(() => ({}))

      if (!r.ok) {
        setError(
          getLoginErrorMessage(d, 'Password reset failed.')
        )
        return
      }

      setDone(true)
    } catch (error) {
      console.error(
        '[IntelliWatch] Password reset failed:',
        error
      )

      setError('Unable to connect to the server. Please try again.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div className="modal-box" onClick={e => e.stopPropagation()}>
        <div className="modal-header">
          <span className="modal-title">Reset Password</span>

          <button
            className="modal-close"
            onClick={onClose}
            aria-label="Close"
            type="button"
          >
            &times;
          </button>
        </div>

        {done ? (
          <div className="modal-success">
            <span className="modal-success-icon">✅</span>
            <p>Your password has been updated successfully.</p>

            <button
              className="login-btn"
              onClick={onClose}
              style={{ marginTop: 12 }}
              type="button"
            >
              Done
            </button>
          </div>
        ) : step === 'request' ? (
          <form className="login-form" onSubmit={handleRequest} noValidate>
            <p className="login-fp-desc">
              Enter your username to request a password reset.
            </p>

            <div className="login-field">
              <label className="login-label" htmlFor="fp-u">
                Username
              </label>

              <div className="login-input-wrap">
                <span className="login-input-icon">👤</span>

                <input
                  id="fp-u"
                  className="login-input"
                  type="text"
                  placeholder="Enter your username"
                  value={username}
                  onChange={e => setUsername(e.target.value)}
                  disabled={loading}
                  spellCheck={false}
                />
              </div>
            </div>

            {error && <div className="login-error">⚠️ {error}</div>}
            {info && <div className="login-info">{info}</div>}

            <button className="login-btn" type="submit" disabled={loading}>
              {loading ? (
                <span className="login-spinner" />
              ) : (
                'Get Reset Token'
              )}
            </button>
          </form>
        ) : (
          <form className="login-form" onSubmit={handleReset} noValidate>
            {info && (
              <div className="login-info login-info--token">
                <pre>{info}</pre>
              </div>
            )}

            <div className="login-field">
              <label className="login-label" htmlFor="fp-tok">
                Reset Token
              </label>

              <div className="login-input-wrap">
                <span className="login-input-icon">🎫</span>

                <input
                  id="fp-tok"
                  className="login-input"
                  type="text"
                  placeholder="Paste your reset token"
                  value={token}
                  onChange={e => setToken(e.target.value)}
                  disabled={loading}
                  spellCheck={false}
                />
              </div>
            </div>

            <div className="login-field">
              <label className="login-label" htmlFor="fp-pw">
                New Password
              </label>

              <div className="login-input-wrap">
                <span className="login-input-icon">🔒</span>

                <input
                  id="fp-pw"
                  className="login-input"
                  type={showPw ? 'text' : 'password'}
                  placeholder="Enter your new password"
                  value={newPw}
                  onChange={e => setNewPw(e.target.value)}
                  autoComplete="new-password"
                  disabled={loading}
                />

                <button

                  onClick={() => setShowPw(v => !v)}
                  aria-label={showPw ? 'Hide password' : 'Show password'}

            <div className="login-field">
              <label className="login-label" htmlFor="fp-pw2">

              </label>

              <div className="login-input-wrap">
                <span className="login-input-icon">🔒</span>

                <input

                  className="login-input"
                  type={showPw ? 'text' : 'password'}
                  placeholder="Re-enter your password"

export default function LoginPage({ onLogin }) {
  useEffect(() => {
    try {
      localStorage.removeItem('iw_saved_accounts')
    } catch (error) {
      console.warn('[IntelliWatch] Legacy saved-account history could not be cleared:', error)
    }
  }, [])

  const [mode, setMode] = useState('operator')
  const [view, setView] = useState('login')
  const [activeRole, setActiveRole] = useState('OPERATOR')

  const [username, setUsername] = useState('')
  const [password, setPassword] = useState('')
  const [showPass, setShowPass] = useState(false)

  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)
  const [modal, setModal] = useState(null)

  const [availability, setAvailability] = useState({
    checked: false,
    internetAvailable: false,
    backendAvailable: false,
    onlineMode: false,
    registrationEnabled: false,
  })

  const onlineMode = availability.onlineMode
  const regEnabled =
    onlineMode && availability.registrationEnabled


  // ==================================================
  // AUTOMATIC AVAILABILITY DETECTION
  // ==================================================

  useEffect(() => {
    let cancelled = false

    async function updateAvailability() {
      const result = await detectAvailability()

      if (!cancelled) {
        setAvailability(result)

        if (!result.onlineMode) {
          setModal(null)

          if (result.registrationEnabled === false) {
            setView('login')
          }
        }
      }
    }

    function handleOffline() {
      updateAvailability()
    }

    function handleOnline() {
      updateAvailability()
    }

    updateAvailability()

    window.addEventListener('online', handleOnline)
    window.addEventListener('offline', handleOffline)

    return () => {

========== LOGIN COMPONENT PART 4 ==========
      cancelled = true

      window.removeEventListener('online', handleOnline)
      window.removeEventListener('offline', handleOffline)
    }
  }, [])


  // ==================================================
  // DEMO LOGIN
  //
  // All three roles are available in fallback mode.
  // The selected tab determines the simulated role.
  // ==================================================

  function completeDemoLogin(requestedRole) {
    const demoUser = {
      access_token: DEMO_TOKEN,
      username: 'demo',
      full_name: 'Demo User',
      role: requestedRole,
      account_role: 'DEMO',
      is_restricted_session: true,
      is_demo_session: true,
      user_id: 0,
      session_mode: 'DEMO',
    }

    const sessionUser = {
      username: demoUser.username,
      full_name: demoUser.full_name,
      role: demoUser.role,
      account_role: demoUser.account_role,
      is_restricted_session: true,
      is_demo_session: true,
      user_id: demoUser.user_id,
      session_mode: 'DEMO',
    }

    try {
      saveSession(DEMO_TOKEN, sessionUser)
      onLogin(demoUser)
    } catch (error) {
      clearSavedSession()

      console.error(
        '[IntelliWatch] Demo session initialization failed:',
        error
      )

      setError(
        'The demo session could not be initialized. Please allow session storage and refresh the application.'
      )
    }
  }


  // ==================================================
  // ROLE SWITCHING
  // ==================================================

  function switchMode(nextMode) {
    setMode(nextMode)
    setUsername('')
    setPassword('')
    setError('')
  }

  function switchActiveRole(nextRole) {
    setActiveRole(nextRole)
    setError('')
  }



  // ==================================================
  // LOGIN
  // ==================================================

  async function handleSubmit(e) {
    e.preventDefault()

    if (loading) return

    setError('')

    const enteredUsername = username.trim()

    if (!enteredUsername || !password) {
      setError('Please enter both username and password.')
      return
    }

    const requestedRole =
      mode === 'admin'
        ? 'ADMIN'
        : activeRole

    const exactDemoCredentials =
      DEMO_MODE &&
      enteredUsername.toLowerCase() === 'demo' &&
      password === 'demo'

    setLoading(true)

    try {
      // ------------------------------------------------
      // 1. DEMO LOGIN
      // ------------------------------------------------
      // Demo credentials create a local, restricted session.
      // They never authenticate against the real backend.

      

      // ------------------------------------------------
      // 2. CHECK SERVICE AVAILABILITY
      // ------------------------------------------------

      const currentAvailability = await detectAvailability()

      setAvailability(currentAvailability)

      // ------------------------------------------------
      // 3. REGISTERED ACCOUNT AUTHENTICATION
      // ------------------------------------------------

      if (!currentAvailability.backendAvailable) {
        if (exactDemoCredentials) {
          completeDemoLogin(requestedRole)
          return
        }
        setError(
          DEMO_MODE
            ? 'Online services are unavailable. Use demo / demo for the local demonstration.'
            : 'The IntelliWatch backend is unavailable. Start the backend and try again.'
        )
        return
      }
      // ------------------------------------------------

      let response

      try {
        response = await fetch(`${API}/auth/login`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            username: enteredUsername,
            password,
            requested_role: requestedRole,
          }),
        })
      } catch (networkError) {
        console.error(
          '[IntelliWatch Login] Authentication request failed:',
          networkError
        )

        const refreshedAvailability = await detectAvailability()

        setAvailability(refreshedAvailability)

        if (exactDemoCredentials && !refreshedAvailability.backendAvailable) {
          completeDemoLogin(requestedRole)
          return
        }

        setError(
          refreshedAvailability.backendAvailable
            ? 'Unable to reach the authentication endpoint. Check the HTTPS certificate, API URL and CORS configuration.'
            : 'The IntelliWatch backend is unavailable. Please start the backend and try again.'
        )

        return
      }

      // ------------------------------------------------
      // 4. PARSE THE SERVER RESPONSE
      // ------------------------------------------------

      if (!response.ok && [502, 503, 504].includes(response.status) && exactDemoCredentials) {
        completeDemoLogin(requestedRole)
        return
      }

      const responseText = await response.text()

      let data = {}

      if (responseText.trim()) {
        try {
          data = JSON.parse(responseText)
        } catch (parseError) {
          console.error(
            '[IntelliWatch Login] Invalid JSON response:',
            parseError
          )

          setError(
            `The authentication server returned an invalid response (HTTP ${response.status}).`
          )

          return
        }
      }

      if (
        !data ||
        typeof data !== 'object' ||
        Array.isArray(data)
      ) {
        console.error(
          '[IntelliWatch Login] Server returned an invalid response object.'
        )

        setError(
          'The authentication server returned an invalid response.'
        )

        return
      }

      // ------------------------------------------------
      // 5. HANDLE BACKEND AUTHENTICATION ERRORS
      // ------------------------------------------------

      if (!response.ok) {
        const serverMessage = getLoginErrorMessage(
          data,
          `Authentication failed (HTTP ${response.status}).`
        )

        console.warn(
          '[IntelliWatch Login] Backend rejected authentication:',
          {
            status: response.status,
            message: serverMessage,
          }
        )

        setError(serverMessage)
        return
      }

      // ------------------------------------------------
      // 6. VALIDATE SUCCESSFUL AUTHENTICATION RESPONSE
      // ------------------------------------------------

      if (
        typeof data.access_token !== 'string' ||
        !data.access_token.trim()
      ) {
        console.error(
          '[IntelliWatch Login] Access token missing from successful response.'
        )

        setError(
          'The authentication server did not return a valid access token.'
        )

        return
      }

      if (
        typeof data.username !== 'string' ||
        typeof data.role !== 'string' ||
        !['ADMIN', 'OPERATOR', 'MAINTENANCE_ENGINEER'].includes(data.role) ||
        data.role !== requestedRole ||
        !Number.isInteger(data.user_id) ||
        typeof data.full_name !== 'string'
      ) {
        console.error(
          '[IntelliWatch Login] Incomplete or invalid account information returned by the backend.'
        )

        setError(
          'The authentication server returned incomplete account information.'
        )

        return
      }

      const onlineUser = {
        ...data,
        account_role: data.account_role || data.role,
        is_restricted_session: !!data.is_restricted_session,
        is_demo_session: false,
        session_mode: 'ONLINE',
      }

      const sessionUser = {
        username: data.username,
        full_name: data.full_name,
        role: data.role,
        account_role: data.account_role || data.role,
        is_restricted_session: !!data.is_restricted_session,
        is_demo_session: false,
        user_id: data.user_id,
        session_mode: 'ONLINE',
      }

      // ------------------------------------------------
      // 7. STORE THE AUTHENTICATED SESSION
      // ------------------------------------------------

      try {
        saveSession(data.access_token, sessionUser)
      } catch (storageError) {
        console.error(
          '[IntelliWatch Login] Session storage failed:',
          storageError
        )

        clearSavedSession()

        setError(
          'Your browser could not store the login session. Please enable session storage and try again.'
        )

        return
      }

      // ------------------------------------------------
      // 9. INITIALIZE THE AUTHENTICATED APPLICATION
      // ------------------------------------------------

      try {
        onLogin(onlineUser)
      } catch (applicationError) {
        console.error(
          '[IntelliWatch Login] Application initialization failed after authentication:',
          applicationError
        )

        clearSavedSession()

        setError(
          'Authentication succeeded, but the application could not initialize. Refresh the page and inspect the application console.'
        )
      }

    } catch (unexpectedError) {
      console.error(
        '[IntelliWatch Login] Unexpected sign-in failure:',
        unexpectedError
      )

      setError(
        'An unexpected sign-in error occurred. Check the browser console for details.'
      )
    } finally {
      setLoading(false)
    }
  }


  // ==================================================
  // CREATE ACCOUNT VIEW
  // ==================================================

  if (view === 'create') {
    if (!regEnabled) {
      return (
        <div className="login-root">
          <div className="login-card">
            <div className="login-error" role="alert">
              Account registration requires both internet and an available
              IntelliWatch backend.
            </div>

            <button
              type="button"
              className="login-btn"
              onClick={() => setView('login')}
            >
              ← Back to Sign In
            </button>
          </div>
        </div>
      )
    }

    return (
      <CreateAccountView
        onBackToLogin={() => setView('login')}
      />
    )
  }


  // ==================================================
  // LOGIN PAGE UI
  // ==================================================

  return (
    <div className="login-root">
      <div className="login-bg">
        <span className="orb orb-1" />
        <span className="orb orb-2" />
        <span className="orb orb-3" />
      </div>

      <div className="login-stack">
        <div className="login-card">
          <div className="login-brand">
            <span className="login-eye" aria-hidden="true">
              &#128065;&#65039;
            </span>

            <div>
              <h1 className="login-title">
                IntelliWatch
              </h1>

              <p className="login-sub">
                Real-Time Monitoring & Decision Support
              </p>
            </div>
          </div>


          {/* ROLE MODE SELECTOR */}

          <div className="mode-selector">
            <button
              type="button"
              className={`mode-tab${mode === 'operator' ? ' mode-tab--active' : ''}`}
              onClick={() => switchMode('operator')}
            >
              <span className="mode-tab-icon">📊</span>

              <span className="mode-tab-text">
                <span className="mode-tab-title">
                  Operator / Engineer
                </span>

                <span className="mode-tab-sub">
                  Equipment monitoring and maintenance
                </span>
              </span>
            </button>

            <button
              type="button"
              className={`mode-tab mode-tab--admin${mode === 'admin' ? ' mode-tab--active-admin' : ''}`}

            >
              <span className="mode-tab-icon">🛡️</span>

========== LOGIN COMPONENT PART 5 ==========

              <span className="mode-tab-text">
